\begin{frame}{Сравнения по модулю}
  \begin{definition}
    $a$ и $b$ сравнимы по модулю $m$, если $m \mid (a-b)$: \quad $\congr{a}{b}{m}$.
  \end{definition}
  \notearea
\end{frame}

\begin{theoremframe}{Малая теорема Ферма}
  Если $p$ "--- простое и $p \nmid a$, то $\congr{a^{p-1}}{1}{p}$.
\end{theoremframe}
