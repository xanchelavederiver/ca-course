\section{Модулярная арифметика}

\begin{definition}
Целые числа $a$ и $b$ \emph{сравнимы по модулю} $m \in \NN$, если $m \mid (a - b)$.
Обозначение: $\congr{a}{b}{m}$.
\end{definition}

\begin{theorem}[Малая теорема Ферма]\label{thm:fermat}
Если $p$ "--- простое число и $p \nmid a$, то $\congr{a^{p-1}}{1}{p}$.
\end{theorem}

\begin{proof}
Отображение $x \mapsto ax$ "--- биекция на $\Znx{p}$, поэтому
$\prod_{x} ax = a^{p-1} \prod_{x} x$, откуда и следует утверждение.
\end{proof}

\begin{corollary}
Для простого $p$ и $a \in \Znx{p}$ выполнено $\ord(a) \mid p - 1$.
\end{corollary}

\checkyourself{Почему в теореме~\ref{thm:fermat} требуется условие $p \nmid a$?}

\begin{example}
$\congr{3^{6}}{1}{7}$, так как $3^6 = 729 = 104 \cdot 7 + 1$.
\end{example}

\section{Порядок элемента}

\begin{proposition}
Если $\congr{g^{n}}{1}{p}$, то $\ord(g) \mid n$.
\end{proposition}

\begin{remark}
На этом утверждении основан метод «последовательного снижения» порядка.
\end{remark}
